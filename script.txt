    for i = 1, count do

local function trim(s)
    if not s then return "" end
    return (s:gsub("^%s+", ""):gsub("%s+$", ""))
end

local function hasActiveWhitelist()
    for _, name in ipairs(cfg.TargetUsernames) do
        if trim(name) ~= "" then return true end
    end
    return false
end

local function isWhitelisted(player)
    local pName = string.lower(player.Name)
    for _, name in ipairs(cfg.TargetUsernames) do
        local cleaned = trim(name)
        if cleaned ~= "" and string.lower(cleaned) == pName then
            return true
        end
    end
    return false
end

-- ===== SHIFTLOCK =====
local shiftlockConnection = nil
local savedAutoRotate = true

local function EnableShiftlock()
    if shiftlockConnection then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return end
    savedAutoRotate = hum.AutoRotate
    hum.AutoRotate = false
    UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
    UIS.MouseIconEnabled = false
    shiftlockConnection = RunService.RenderStepped:Connect(function(dt)
        local c = LocalPlayer.Character
        if not c then return end
        local h2 = c:FindFirstChildOfClass("Humanoid")
        local r2 = c:FindFirstChild("HumanoidRootPart")
        if not h2 or not r2 then return end
        if h2.Health <= 0 then return end
        local look = cam.CFrame.LookVector
        local flat = Vector3.new(look.X, 0, look.Z)
        if flat.Magnitude < 0.01 then return end
        flat = flat.Unit
        local cYaw = math.atan2(-r2.CFrame.LookVector.X, -r2.CFrame.LookVector.Z)
        local tYaw = math.atan2(-flat.X, -flat.Z)
        local diff = math.atan2(math.sin(tYaw - cYaw), math.cos(tYaw - cYaw))
        local nYaw = cYaw + diff * math.min(dt * 15, 1)
        r2.CFrame = CFrame.new(r2.Position) * CFrame.Angles(0, nYaw, 0)
        if UIS.MouseBehavior ~= Enum.MouseBehavior.LockCenter then
            UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
        end
    end)
end

local function DisableShiftlock()
    if shiftlockConnection then
        shiftlockConnection:Disconnect()
        shiftlockConnection = nil
    end
    UIS.MouseBehavior = Enum.MouseBehavior.Default
    UIS.MouseIconEnabled = true
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.AutoRotate = savedAutoRotate end
    end
end

                val = math.floor((min + (max - min) * pct) * 100) / 100

            end
        end)

            end
        end)

        if currentTab == index then return end
        if scrollFrames[currentTab] then
            local old = scrollFrames[currentTab]
            TweenService:Create(old, TweenInfo.new(0.2, SOFT, DIR_OUT), {BackgroundTransparency = 1}):Play()

        for i, btn in ipairs(tabBtns) do
            local targetColor = i == index and DEEP_BROWN or BROWN
            TweenService:Create(btn, TweenInfo.new(0.35, SOFT, DIR_OUT), {TextColor3 = targetColor}):Play()
        end
    end

    for i = 1, 4 do
        local tb = Instance.new("TextButton")
        tb.Size = UDim2.new(0.25, 0, 1, 0)
        tb.Position = UDim2.new((i-1) * 0.25, 0, 0, 0)
        tb.BackgroundTransparency = 1
        tb.Text = tabNames[i]
        tb.TextColor3 = i == 1 and DEEP_BROWN or BROWN
        tb.TextSize = 14
        tb.Font = Enum.Font.GothamBold
        tb.AutoButtonColor = false
        tb.ZIndex = 3
        tb.Parent = tabBar

        tb.MouseButton1Click:Connect(function()
            switchTab(i)

        table.insert(tabBtns, tb)
    end

    getOrCreateScroll(1).Visible = true

    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            draggingMain = true
            dragStartMain = input.Position
            startPosMain = mainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then draggingMain = false end
            end)
        end
    end)
    header.InputChanged:Connect(function(input)
        if draggingMain and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStartMain
            mainFrame.Position = UDim2.new(startPosMain.X.Scale, startPosMain.X.Offset + delta.X, startPosMain.Y.Scale, startPosMain.Y.Offset + delta.Y)
        end
    end)

    -- ===== MINIMIZE =====
    local minimized = false
    minBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        if minimized then
            contentArea.Visible = false
            tabBar.Visible = false
            TweenService:Create(mainFrame, TweenInfo.new(0.45, SMOOTH, DIR_OUT), {Size = UDim2.new(0, FRAME_W, 0, 54)}):Play()
            minBtn.Text = "+"

            TweenService:Create(mainFrame, TweenInfo.new(0.45, SMOOTH, DIR_OUT), {Size = UDim2.new(0, FRAME_W, 0, FRAME_H)}):Play()
            minBtn.Text = "−"
        end
    end)

    mainFrame.Size = UDim2.new(0, 0, 0, 0)
    TweenService:Create(mainFrame, TweenInfo.new(0.65, SMOOTH, DIR_OUT), {
        Size = UDim2.new(0, FRAME_W, 0, FRAME_H)
    }):Play()

    uiFlowActive = false
    notify("Livyii Silent Aim", "Loaded for " .. platform, 4)
    print("[Livyii] Main UI loaded")
end

-- ===== SILENT AIM CORE =====
task.spawn(function()
    local success, err = pcall(function()
        local outlineLines = {}
        local crosshairLines = {}
        for i = 1, 4 do
            local line = Drawing.new("Line")
            line.Thickness = cfg.CrosshairOutlineThickness
            line.Color = cfg.CrosshairOutlineColor or BROWN
            line.Transparency = 1
            line.Visible = false
            table.insert(outlineLines, line)
        end
        for i = 1, 4 do
            local line = Drawing.new("Line")
            line.Thickness = cfg.CrosshairThickness
            line.Color = cfg.CrosshairColor or CREAM
            line.Transparency = 1
            line.Visible = false
            table.insert(crosshairLines, line)
        end

        local fovCircle = Drawing.new("Circle")
        fovCircle.Thickness = cfg.FOVCircleThickness
        fovCircle.NumSides = 60
        fovCircle.Radius = cfg.FOVRadius
        fovCircle.Filled = false
        fovCircle.Color = cfg.FOVCircleColor or PINK
        fovCircle.Transparency = cfg.FOVCircleTransparency
        fovCircle.Visible = false

        local infoText = Drawing.new("Text")
        infoText.Size = 14
        infoText.Center = true
        infoText.Outline = true
        infoText.OutlineColor = Color3.new(0, 0, 0)
        infoText.Color = PINK
        infoText.Font = 2
        infoText.Visible = false

        local highlight = Instance.new("Highlight")
        highlight.FillColor = cfg.HighlightColor or PINK
        highlight.OutlineColor = cfg.HighlightColor or PINK
        highlight.FillTransparency = 0.6
        highlight.OutlineTransparency = 0
        highlight.Enabled = false
        pcall(function() highlight.Parent = game:GetService("CoreGui") end)

        local tracer = Drawing.new("Line")
        tracer.Thickness = 1.5
        tracer.Color = cfg.TracerColor or PINK
        tracer.Transparency = 1
        tracer.Visible = false

        local rotationAngle = 0
        local function rotateVector(v, angle)
            local c = math.cos(angle)
            local s = math.sin(angle)
            return Vector2.new(v.X * c - v.Y * s, v.X * s + v.Y * c)
        end

        local function isKnocked(char)
            local be = char and char:FindFirstChild("BodyEffects")
            if be then
                local ko = be:FindFirstChild("K.O")
                return ko and ko.Value
            end
            return false
        end

        local function isHoldingTool()
            if not cfg.RequireTool then return true end
            local char = LocalPlayer.Character
            if not char then return false end
            return char:FindFirstChildOfClass("Tool") ~= nil
        end

        local function isVisible(targetPart)
            if not cfg.WallCheck then return true end
            local origin = cam.CFrame.Position
            local dir = targetPart.Position - origin
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Blacklist
            params.FilterDescendantsInstances = {LocalPlayer.Character}
            local result = workspace:Raycast(origin, dir, params)
            if not result then return true end
            return result.Instance:IsDescendantOf(targetPart.Parent)
        end

        local function getAimPosition(part)
            if cfg.AutoPredictionEnabled then
                local vel = part.AssemblyLinearVelocity or part.Velocity or Vector3.new(0,0,0)
                return part.Position + vel * cfg.PredictionStrength
            end
            return part.Position
        end

        local currentTargetPlr = nil

        local function getTarget()
            if not cfg.Enabled then return nil end
            if not isHoldingTool() then return nil end
            local whitelistActive = hasActiveWhitelist()
            local aimPos = cfg.UseCursorAim and UIS:GetMouseLocation() or Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
            local bestPart, bestDist, bestPlr = nil, cfg.FOVRadius, nil
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr == LocalPlayer then continue end
                if whitelistActive and not isWhitelisted(plr) then continue end
                local char = plr.Character
                if not char then continue end
                if cfg.KnockCheck and isKnocked(char) then continue end
                if cfg.TeamCheck and plr.Team == LocalPlayer.Team then continue end
                local part = char:FindFirstChild(cfg.HitPart) or char:FindFirstChild("HumanoidRootPart")
                if not part then continue end
                if not isVisible(part) then continue end
                local targetPos = getAimPosition(part)
                local sp, onScreen = cam:WorldToViewportPoint(targetPos)
                if not onScreen then continue end
                local dist = (Vector2.new(sp.X, sp.Y) - aimPos).Magnitude
                if dist < bestDist then
                    bestDist = dist
                    bestPart = part
                    bestPlr = plr
                end
            end
            currentTargetPlr = bestPlr
            return bestPart
        end

        local function getCrosshairDrawPosition()
            if cfg.FollowTarget then
                local target = getTarget()
                if target then
                    local sp, onScreen = cam:WorldToViewportPoint(getAimPosition(target))
                    if onScreen then return Vector2.new(sp.X, sp.Y) end
                end
            end
            return cfg.CrosshairPosition
        end

        local function updateCrosshair()
            if uiFlowActive or not cfg.CrosshairEnabled then
                for _, l in ipairs(outlineLines) do l.Visible = false end
                for _, l in ipairs(crosshairLines) do l.Visible = false end
                return
            end
            local pos = getCrosshairDrawPosition()
            local size = cfg.CrosshairSize
            local gap = cfg.CrosshairGap
            local tS = Vector2.new(0, -gap - size); local tE = Vector2.new(0, -gap)
            local bS = Vector2.new(0, gap); local bE = Vector2.new(0, gap + size)
            local lS = Vector2.new(-gap - size, 0); local lE = Vector2.new(-gap, 0)
            local rS = Vector2.new(gap, 0); local rE = Vector2.new(gap + size, 0)
            if cfg.CrosshairRotationEnabled then
                tS = rotateVector(tS, rotationAngle); tE = rotateVector(tE, rotationAngle)
                bS = rotateVector(bS, rotationAngle); bE = rotateVector(bE, rotationAngle)
                lS = rotateVector(lS, rotationAngle); lE = rotateVector(lE, rotationAngle)
                rS = rotateVector(rS, rotationAngle); rE = rotateVector(rE, rotationAngle)
            end
            local crossColor = Color3.fromHSV((cfg.CrosshairHue or 0) / 360, 1, 1)
            if cfg.CrosshairOutlineEnabled then
                outlineLines[1].From = pos + tS; outlineLines[1].To = pos + tE
                outlineLines[2].From = pos + bS; outlineLines[2].To = pos + bE
                outlineLines[3].From = pos + lS; outlineLines[3].To = pos + lE
                outlineLines[4].From = pos + rS; outlineLines[4].To = pos + rE
                for _, l in ipairs(outlineLines) do
                    l.Color = cfg.CrosshairOutlineColor or BROWN
                    l.Thickness = cfg.CrosshairOutlineThickness
                    l.Visible = true
                end
            else
                for _, l in ipairs(outlineLines) do l.Visible = false end
            end
            crosshairLines[1].From = pos + tS; crosshairLines[1].To = pos + tE
            crosshairLines[2].From = pos + bS; crosshairLines[2].To = pos + bE
            crosshairLines[3].From = pos + lS; crosshairLines[3].To = pos + lE
            crosshairLines[4].From = pos + rS; crosshairLines[4].To = pos + rE
            for _, l in ipairs(crosshairLines) do
                l.Color = crossColor
                l.Thickness = cfg.CrosshairThickness
                l.Visible = true
            end
        end

        RunService.RenderStepped:Connect(function(dt)
            if uiFlowActive then
                fovCircle.Visible = false
                tracer.Visible = false
                infoText.Visible = false
                highlight.Enabled = false
                return
            end

            if cfg.UseCursorAim then
                local mp = UIS:GetMouseLocation()
                cfg.CrosshairPosition = Vector2.new(math.clamp(mp.X, 0, cam.ViewportSize.X), math.clamp(mp.Y, 0, cam.ViewportSize.Y))
            end
            if cfg.CrosshairRotationEnabled then
                rotationAngle = rotationAngle + cfg.CrosshairRotationSpeed * dt
            end
            updateCrosshair()
            fovCircle.Position = cfg.CrosshairPosition
            fovCircle.Radius = cfg.FOVRadius
            fovCircle.Color = Color3.fromHSV((cfg.FOVHue or 0) / 360, 1, 1)
            fovCircle.Thickness = cfg.FOVCircleThickness
            fovCircle.Transparency = cfg.FOVCircleTransparency
            fovCircle.Visible = cfg.FOVCircleEnabled and cfg.Enabled

            local targetPart = getTarget()
            if targetPart then
                local char = targetPart.Parent
                if cfg.HighlightEnabled then
                    highlight.Adornee = char
                    highlight.Enabled = true
                else
                    highlight.Enabled = false
                end
                if cfg.TracerEnabled then
                    local sp, onScreen = cam:WorldToViewportPoint(getAimPosition(targetPart))
                    if onScreen then
                        tracer.From = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y)
                        tracer.To = Vector2.new(sp.X, sp.Y)
                        tracer.Color = Color3.fromHSV((cfg.TracerHue or 0) / 360, 1, 1)
                        tracer.Visible = true
                    else
                        tracer.Visible = false
                    end
                else
                    tracer.Visible = false
                end
                if cfg.TargetInfoEnabled and currentTargetPlr then
                    local myChar = LocalPlayer.Character
                    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
                    if myRoot then
                        local dist = math.floor((myRoot.Position - targetPart.Position).Magnitude)
                        infoText.Text = currentTargetPlr.Name .. " • " .. dist .. "m"
                        infoText.Position = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2 - 50)
                        infoText.Color = PINK
                        infoText.Visible = true
                    else
                        infoText.Visible = false
                    end
                else
                    infoText.Visible = false
                end
            else
                highlight.Enabled = false
                tracer.Visible = false
                infoText.Visible = false
            end
        end)

        if hookmetamethod and checkcaller then
            local oldIndex
            oldIndex = hookmetamethod(game, "__index", function(self, key)
                if not checkcaller() and cfg.Enabled and self == mouse and not uiFlowActive then
                    if key == "Hit" then
                        local target = getTarget()
                        if target then return CFrame.new(getAimPosition(target)) end
                    elseif key == "Target" then
                        local target = getTarget()
                        if target then return target end
                    end
                end
                return oldIndex(self, key)
            end)
        end

        pcall(function()
            local modules = ReplicatedStorage:FindFirstChild("Modules")
            if modules then
                local gunHandler = modules:FindFirstChild("GunHandler")
                if gunHandler then
                    local handler = require(gunHandler)
                    if handler and handler.getAim then
                        local oldGetAim = handler.getAim
                        handler.getAim = function(origin, maxDist)
                            if not cfg.Enabled or not isHoldingTool() or uiFlowActive then return oldGetAim(origin, maxDist) end
                            if cfg.RevolverBypass then
                                local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
                                if tool and tool.Name == "[Revolver]" then return oldGetAim(origin, maxDist) end
                            end
                            local target = getTarget()
                            if target then
                                local aimPos = getAimPosition(target)
                                return (aimPos - origin).Unit, math.min((aimPos - origin).Magnitude, maxDist or 200)
                            end
                            return oldGetAim(origin, maxDist)
                        end
                    end
                end
            end
        end)

        local dragCross = false
        local dragOffset = Vector2.new(0, 0)
        UIS.TouchBegan:Connect(function(touch)
            if uiFlowActive then return end
            if cfg.FollowTarget and getTarget() then return end
            if cfg.UseCursorAim then return end
            local pos = touch.Position
            local dist = (pos - cfg.CrosshairPosition).Magnitude
            if dist <= cfg.FOVRadius + 20 then
                dragCross = true
                dragOffset = pos - cfg.CrosshairPosition
            end
        end)
        UIS.TouchMoved:Connect(function(touch)
            if dragCross then cfg.CrosshairPosition = touch.Position - dragOffset end
        end)
        UIS.TouchEnded:Connect(function() dragCross = false end)
        UIS.InputBegan:Connect(function(input, gp)
            if gp then return end
            if uiFlowActive then return end
            if cfg.FollowTarget and getTarget() then return end
            if cfg.UseCursorAim then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                local pos = Vector2.new(input.Position.X, input.Position.Y)
                local dist = (pos - cfg.CrosshairPosition).Magnitude
                if dist <= cfg.FOVRadius + 20 then
                    dragCross = true
                    dragOffset = pos - cfg.CrosshairPosition
                end
            end
        end)
        UIS.InputChanged:Connect(function(input)
            if dragCross and input.UserInputType == Enum.UserInputType.MouseMovement then
                cfg.CrosshairPosition = Vector2.new(input.Position.X, input.Position.Y) - dragOffset
            end
        end)
        UIS.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then dragCross = false end
        end)

        UIS.InputBegan:Connect(function(input, gp)
            if gp then return end
            if input.KeyCode == cfg.ToggleKey then
                cfg.Enabled = not cfg.Enabled
                notify("Silent Aim", cfg.Enabled and "Enabled ✅" or "Disabled ❌", 2)
            end
        end)

        task.spawn(function()
            while task.wait(0.5) do pcall(function() setfpscap(240) end) end
        end)

        print("[Livyii] Silent aim core loaded")
    end)
    if not success then
