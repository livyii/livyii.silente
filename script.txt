-- // 🎯 SIMPLE CURSOR TRIGGERBOT LOGIC 🎯 //
local simple_triggerbot = {}
simple_triggerbot.is_firing = false

simple_triggerbot.get_gun = function()
    local char = LocalPlayer.Character
    if not char then return nil end
    for _, tool in next, char:GetChildren() do
        if tool:IsA("Tool") and tool:FindFirstChild("Ammo") then return tool end
    end
    return nil
end

simple_triggerbot.rapid = function(tool)
    tool:Activate()
end

local function startFiring()
    if not cfg.CursorTriggerbotEnabled then return end
    if simple_triggerbot.is_firing then return end
    local gun = simple_triggerbot.get_gun()
    if not gun then return end
    simple_triggerbot.is_firing = true
    task.spawn(function()
        while simple_triggerbot.is_firing do
            local g = simple_triggerbot.get_gun()
            if not g or not cfg.CursorTriggerbotEnabled then break end
            simple_triggerbot.rapid(g)
            task.wait(cfg.CursorTriggerbotDelay)
        end
        simple_triggerbot.is_firing = false
    end)
end

local function stopFiring()
    simple_triggerbot.is_firing = false
end

-- // SHIFTLOCK LOGIC //
local shiftlock_button_cache = nil
local function find_shiftlock_button()
    if shiftlock_button_cache and shiftlock_button_cache.Parent then return shiftlock_button_cache end
    shiftlock_button_cache = nil
    local search_roots = { GuiParent, LocalPlayer:FindFirstChild("PlayerGui") }
    for _, root in ipairs(search_roots) do
        if root then
            local ok, descs = pcall(function() return root:GetDescendants() end)
            if ok and descs then
                for _, d in ipairs(descs) do
                    if d:IsA("GuiButton") then
                        local n = d.Name:lower()
                        if n:find("shiftlock") or n:find("shift_lock") or n:find("shift lock") or n == "shift" or n == "mouselock" or n == "lockbutton" then
                            shiftlock_button_cache = d
                            return d
                        end
                    end
                end
            end
        end
    end
    return nil
end

local function press_button(btn)
    pcall(function() btn:Fire("Activated") end)
    pcall(function() btn:Fire("MouseButton1Click") end)
    pcall(function() if firetouchinterest then firetouchinterest(btn, Enum.UserInputType.Touch, 0); task.wait(0.05); firetouchinterest(btn, Enum.UserInputType.Touch, 1) end end)
end

local last_shiftlock_toggle = 0
local shiftlock_ui_refresh = nil

local function try_toggle_shiftlock()
    if not cfg.RightClickShiftlock then return end
    local now = tick()
    if now - last_shiftlock_toggle < 0.25 then return end
    last_shiftlock_toggle = now
    local btn = find_shiftlock_button()
    if btn then
        press_button(btn)
        cfg.ShiftlockEnabled = not cfg.ShiftlockEnabled
        if shiftlock_ui_refresh then pcall(shiftlock_ui_refresh, cfg.ShiftlockEnabled) end
        notify("🍮 shiftlock", cfg.ShiftlockEnabled and "on" or "off", 1.5)
        return
    end
    local new_state = not cfg.ShiftlockEnabled
    cfg.ShiftlockEnabled = new_state
    pcall(function()
        local svc = UserSettings():GetService("UserGameSettings")
        svc.RotationType = new_state and Enum.RotationType.MovementRelative or Enum.RotationType.CameraRelative
    end)
    if shiftlock_ui_refresh then pcall(shiftlock_ui_refresh, new_state) end
    notify("🍮 shiftlock", new_state and "on" or "off", 1.5)
end

end

local function is_touch_on_ui(pos)
    if not ui_main_frame or not ui_main_frame.Visible then return false end

end

-- ==========================================
-- // 🎯 CORE ENGINE 🎯 //
-- ==========================================
task.spawn(function()
    local ok, err = pcall(function()
        local outline_lines = {}
        local crosshair_lines = {}
        
        for i = 1, 4 do
            local line = Drawing.new("Line")
            line.Thickness = cfg.CrosshairOutlineThickness
            line.Color = Color3.fromRGB(10, 10, 15)
            line.Visible = false
            table.insert(outline_lines, line)
            
            local line2 = Drawing.new("Line")
            line2.Thickness = cfg.CrosshairThickness
            line2.Color = GetColor("Accent")
            line2.Visible = false
            table.insert(crosshair_lines, line2)
        end

        local fov_circle = Drawing.new("Circle")
        fov_circle.Thickness = cfg.FOVCircleThickness
        fov_circle.NumSides = 64
        fov_circle.Radius = cfg.FOVRadius
        fov_circle.Filled = false
        fov_circle.Color = GetColor("Accent")
        fov_circle.Visible = false
        
        local info_text = Drawing.new("Text")
        info_text.Size = 14
        info_text.Center = true
        info_text.Outline = true
        info_text.OutlineColor = Color3.new(0, 0, 0)
        info_text.Color = GetColor("TextMain")
        info_text.Font = 2
        info_text.Visible = false
        
        local highlight = Instance.new("Highlight")
        highlight.FillColor = GetColor("Accent")
        highlight.OutlineColor = GetColor("Stroke")
        highlight.FillTransparency = 0.6
        highlight.Enabled = false
        pcall(function() highlight.Parent = game:GetService("CoreGui") end)
        
        local tracer = Drawing.new("Line")
        tracer.Thickness = 1.5
        tracer.Color = GetColor("Accent")
        tracer.Visible = false

        register_cleanup(function()
            for _, l in ipairs(outline_lines) do pcall(function() l:Remove() end) end
            for _, l in ipairs(crosshair_lines) do pcall(function() l:Remove() end) end
            pcall(function() fov_circle:Remove() end)
            pcall(function() info_text:Remove() end)
            pcall(function() tracer:Remove() end)
            pcall(function() highlight:Destroy() end)
        end)

        table.insert(ThemeListeners, function()
            for i = 1, 4 do
                crosshair_lines[i].Color = GetColor("Accent")
            end
            fov_circle.Color = GetColor("Accent")
            info_text.Color = GetColor("TextMain")
            highlight.FillColor = GetColor("Accent")
            highlight.OutlineColor = GetColor("Stroke")
            tracer.Color = GetColor("Accent")
        end)

        local rotation_angle = 0
        local function rotate_vector(v, angle) local c = math.cos(angle); local s = math.sin(angle); return Vector2.new(v.X * c - v.Y * s, v.X * s + v.Y * c) end
        local function is_knocked(char) local be = char and char:FindFirstChild("BodyEffects"); local ko = be and be:FindFirstChild("K.O"); return ko and ko.Value end
        local function is_holding_tool() if not cfg.RequireTool then return true end; local char = LocalPlayer.Character; return char and char:FindFirstChildOfClass("Tool") ~= nil end
        
        local function is_visible(target_part)
            if not cfg.WallCheck then return true end
            local origin = cam.CFrame.Position
            local dir = target_part.Position - origin
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Blacklist
            params.FilterDescendantsInstances = {LocalPlayer.Character}
            local result = workspace:Raycast(origin, dir, params)
            return not result or result.Instance:IsDescendantOf(target_part.Parent)
        end

        local function get_aim_position(part)
            if cfg.AutoPredictionEnabled then
                local vel = part.AssemblyLinearVelocity or part.Velocity or Vector3.new(0,0,0)
                return part.Position + vel * cfg.PredictionStrength
            end
            return part.Position
        end

        local function get_closest_body_part(char, ref_pos)
            local closestPart, shortestDist = nil, math.huge
            for _, pName in ipairs(bodyPartsList) do
                local part = char:FindFirstChild(pName)
                if part then
                    local sp, onScreen = cam:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local dist = (Vector2.new(sp.X, sp.Y) - ref_pos).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestPart = part
                        end
                    end
                end
            end
            return closestPart
        end

        local function get_target_part(char, ref_pos)
            if cfg.SilentAimClosestPart then
                return get_closest_body_part(char, ref_pos) or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
            end
            return char:FindFirstChild(cfg.HitPart) or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
        end

        local current_target_plr = nil
        local function get_target()
            if not cfg.Enabled or not is_holding_tool() then return nil end
            local whitelist_active = has_active_whitelist()
            local ref_pos = get_reference_position()
            local best_part, best_dist, best_plr = nil, cfg.FOVRadius, nil

            for _, plr in ipairs(Players:GetPlayers()) do
                if plr == LocalPlayer or (whitelist_active and not is_whitelisted(plr)) then continue end
                local char = plr.Character
                if not char or (cfg.KnockCheck and is_knocked(char)) or (cfg.TeamCheck and plr.Team == LocalPlayer.Team) then continue end
                
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if hrp and (hrp.Position - cam.CFrame.Position).Magnitude > cfg.MaxDistance then continue end
                
                local part = get_target_part(char, ref_pos)
                if not part or not is_visible(part) then continue end
                
                local sp, on_screen = cam:WorldToViewportPoint(get_aim_position(part))
                if on_screen then
                    local dist = (Vector2.new(sp.X, sp.Y) - ref_pos).Magnitude
                    if dist < best_dist then
                        best_dist = dist
                        best_part = part
                        best_plr = plr
                    end
                end
            end
            current_target_plr = best_plr
            return best_part
        end

        RunService.RenderStepped:Connect(function(dt)
            if ui_flow_active then
                fov_circle.Visible = false
                tracer.Visible = false
                info_text.Visible = false
                highlight.Enabled = false
                return
            end

            cfg.CrosshairPosition = get_reference_position()
            if cfg.CrosshairRotationEnabled then rotation_angle = rotation_angle + cfg.CrosshairRotationSpeed * dt end
            local target_part = get_target()

            if cfg.CrosshairEnabled then
                local pos = (target_part and cfg.FollowTarget) and (select(1, cam:WorldToViewportPoint(get_aim_position(target_part)))) or cfg.CrosshairPosition
                local size = cfg.CrosshairSize
                local gap = cfg.CrosshairGap
                
                local tS, tE = Vector2.new(0, -gap - size), Vector2.new(0, -gap)
                local bS, bE = Vector2.new(0, gap), Vector2.new(0, gap + size)
                local lS, lE = Vector2.new(-gap - size, 0), Vector2.new(-gap, 0)
                local rS, rE = Vector2.new(gap, 0), Vector2.new(gap + size, 0)
                
                if cfg.CrosshairRotationEnabled then
                    tS, tE = rotate_vector(tS, rotation_angle), rotate_vector(tE, rotation_angle)
                    bS, bE = rotate_vector(bS, rotation_angle), rotate_vector(bE, rotation_angle)
                    lS, lE = rotate_vector(lS, rotation_angle), rotate_vector(lE, rotation_angle)
                    rS, rE = rotate_vector(rS, rotation_angle), rotate_vector(rE, rotation_angle)
                end

                outline_lines[1].From, outline_lines[1].To = pos + tS, pos + tE
                outline_lines[2].From, outline_lines[2].To = pos + bS, pos + bE
                outline_lines[3].From, outline_lines[3].To = pos + lS, pos + lE
                outline_lines[4].From, outline_lines[4].To = pos + rS, pos + rE
                
                crosshair_lines[1].From, crosshair_lines[1].To = pos + tS, pos + tE
                crosshair_lines[2].From, crosshair_lines[2].To = pos + bS, pos + bE
                crosshair_lines[3].From, crosshair_lines[3].To = pos + lS, pos + lE
                crosshair_lines[4].From, crosshair_lines[4].To = pos + rS, pos + rE
                
                for i = 1, 4 do
                    outline_lines[i].Thickness = cfg.CrosshairOutlineThickness
                    outline_lines[i].Visible = cfg.CrosshairOutlineEnabled
                    crosshair_lines[i].Thickness = cfg.CrosshairThickness
                    crosshair_lines[i].Visible = true
                end
            else
                for i = 1, 4 do
                    outline_lines[i].Visible = false
                    crosshair_lines[i].Visible = false
                end
            end

            fov_circle.Position = cfg.CrosshairPosition
            fov_circle.Radius = cfg.FOVRadius
            fov_circle.Visible = cfg.FOVCircleEnabled and cfg.Enabled

            if target_part then
                if cfg.HighlightEnabled then
                    highlight.Adornee = target_part.Parent
                    highlight.Enabled = true
                else
                    highlight.Enabled = false
                end
                
                if cfg.TracerEnabled then
                    local sp, on_screen = cam:WorldToViewportPoint(get_aim_position(target_part))
                    if on_screen then
                        tracer.From = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y)
                        tracer.To = Vector2.new(sp.X, sp.Y)
                        tracer.Visible = true
                    else
                        tracer.Visible = false
                    end
                else
                    tracer.Visible = false
                end
                
                if cfg.TargetInfoEnabled and current_target_plr then
                    local my_root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if my_root then
                        info_text.Text = "🍮 " .. current_target_plr.Name .. " • " .. math.floor((my_root.Position - target_part.Position).Magnitude) .. "m"
                        info_text.Position = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2 - 50)
                        info_text.Visible = true
                    else
                        info_text.Visible = false
                    end
                else
                    info_text.Visible = false
                end
            else
                highlight.Enabled = false
                tracer.Visible = false
                info_text.Visible = false
            end
        end)

        if hookmetamethod and checkcaller then
            local old_index
            old_index = hookmetamethod(game, "__index", function(self, key)
                if not checkcaller() and cfg.Enabled and self == mouse and not ui_flow_active then
                    if key == "Hit" then
                        local target = get_target()
                        if target then return CFrame.new(get_aim_position(target)) end
                    elseif key == "Target" then
                        local target = get_target()
                        if target then return target end
                    end
                end
                return old_index(self, key)
            end)
        end

        UIS.InputBegan:Connect(function(input, gp)
            if gp or ui_flow_active then return end
            if input.UserInputType == Enum.UserInputType.MouseButton2 and cfg.RightClickShiftlock then try_toggle_shiftlock() end
            if input.KeyCode == cfg.ToggleKey then
                cfg.Enabled = not cfg.Enabled
                notify("🍮 silent aim", cfg.Enabled and "enabled" or "disabled", 2)
            end
        end)

        local counted_touches = {}
        UIS.TouchStarted:Connect(function(input)
            if ui_flow_active or not cfg.RightClickShiftlock or is_touch_on_ui(input.Position) then return end
            counted_touches[input] = true
            local count = 0
            for _ in pairs(counted_touches) do count = count + 1 end
            if count >= 2 then counted_touches = {}; try_toggle_shiftlock() end
        end)
        UIS.TouchEnded:Connect(function(input) counted_touches[input] = nil end)
        
        task.spawn(function()
            while task.wait(0.5) do pcall(function() setfpscap(240) end) end
        end)
    end)
    if not ok then warn("[livyii] core error: " .. tostring(err)) end
end)

print("[livyii] loaded successfully. Stay cute! 🍮")
