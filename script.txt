local function spawn_float_particles(parent, emojis, count)
    for i = 1, count do
        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(0, 24, 0, 24)
        lbl.Position = UDim2.new(math.random() * 1, 0, math.random() * 1, 0)
        lbl.BackgroundTransparency = 1
        lbl.Text = emojis[math.random(1, #emojis)]
        lbl.TextSize = math.random(14, 22)
        lbl.TextTransparency = 0.45
        lbl.Font = Enum.Font.Gotham
        lbl.Parent = parent
        task.spawn(function()
            while lbl.Parent do
                local start_y = lbl.Position.Y.Scale
                TweenService:Create(lbl, TweenInfo.new(math.random(4, 8), Enum.EasingStyle.Linear), {
                    Position = UDim2.new(lbl.Position.X.Scale, 0, start_y - 0.15, 0),
                    TextTransparency = 0.85,
                }):Play()
                task.wait(8)
                if not lbl.Parent then break end
                lbl.Position = UDim2.new(math.random() * 1, 0, 1, 0)
                lbl.TextTransparency = 0.45
            end
        end)
    end
end

local shiftlock_conn = nil
local saved_auto_rotate = true

local function enable_shiftlock()
    if shiftlock_conn then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return end
    saved_auto_rotate = hum.AutoRotate
    hum.AutoRotate = false
    UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
    UIS.MouseIconEnabled = false
    shiftlock_conn = RunService.RenderStepped:Connect(function(dt)
        local c = LocalPlayer.Character
        if not c then return end
        local h2 = c:FindFirstChildOfClass("Humanoid")
        local r2 = c:FindFirstChild("HumanoidRootPart")
        if not h2 or not r2 then return end
        if h2.Health <= 0 then return end
        local look = cam.CFrame.LookVector
        local flat = Vector3.new(look.X, 0, look.Z)
        if flat.Magnitude < 0.01 then return end
        flat = flat.Unit
        local c_yaw = math.atan2(-r2.CFrame.LookVector.X, -r2.CFrame.LookVector.Z)
        local t_yaw = math.atan2(-flat.X, -flat.Z)
        local diff = math.atan2(math.sin(t_yaw - c_yaw), math.cos(t_yaw - c_yaw))
        local n_yaw = c_yaw + diff * math.min(dt * 15, 1)
        r2.CFrame = CFrame.new(r2.Position) * CFrame.Angles(0, n_yaw, 0)
        if UIS.MouseBehavior ~= Enum.MouseBehavior.LockCenter then
            UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
        end
    end)
end

local function disable_shiftlock()
    if shiftlock_conn then
        shiftlock_conn:Disconnect()
        shiftlock_conn = nil
    end
    UIS.MouseBehavior = Enum.MouseBehavior.Default
    UIS.MouseIconEnabled = true
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.AutoRotate = saved_auto_rotate end
    end
end

            end
        end)

            end

    make_section(misc_scroll, "✨", "misc")

    make_toggle(misc_scroll, "🔒", "shiftlock", function() return cfg.ShiftlockEnabled end, function(v)
        cfg.ShiftlockEnabled = v
        if v then enable_shiftlock() else disable_shiftlock() end
    end)

    make_section(misc_scroll, "👥", "live targets")

    local note = Instance.new("TextLabel")
    note.Size = UDim2.new(1, 0, 0, 16)
    note.BackgroundTransparency = 1
    note.Text = "tap a player to target only them"
    note.TextColor3 = MUTED_TEXT
    note.TextSize = 10
    note.Font = Enum.Font.GothamMedium
    note.TextXAlignment = Enum.TextXAlignment.Left
    note.Parent = misc_scroll

    player_list.BackgroundTransparency = 1

    pl_layout.Padding = UDim.new(0, 6)
    pl_layout.SortOrder = Enum.SortOrder.LayoutOrder
    pl_layout.Parent = player_list

    local function full_rebuild()
        for _, child in ipairs(player_list:GetChildren()) do
            if child:IsA("TextButton") or child:IsA("TextLabel") then child:Destroy() end
        end
        local sorted = {}
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then table.insert(sorted, plr) end
        end
        table.sort(sorted, function(a, b) return a.Name:lower() < b.Name:lower() end)
        for _, plr in ipairs(sorted) do
            make_player_tile(player_list, plr)
        end
        if #sorted == 0 then
            local empty = Instance.new("TextLabel")
            empty.Size = UDim2.new(1, 0, 0, 30)
            empty.BackgroundTransparency = 1
            empty.Text = "no other players yet 🌸"

            empty.Font = Enum.Font.GothamMedium
            empty.Parent = player_list
        end

    Players.PlayerAdded:Connect(function()
        task.wait(0.3)
        full_rebuild()
    end)
    Players.PlayerRemoving:Connect(function(plr)
        cfg.TargetedPlayers[plr] = nil
        task.wait(0.1)
        full_rebuild()
    end)

    local tab_names = {"🎯", "⚔️", "🎨", "✨"}
    local tab_btns = {}
    local current_tab = 1

    local function switch_tab(index)
        if current_tab == index then return end
        if scroll_frames[current_tab] then scroll_frames[current_tab].Visible = false end
        if not scroll_frames[index] then get_or_create_scroll(index) end
        scroll_frames[index].Visible = true
        current_tab = index
        local target_pos = UDim2.new((index - 1) * 0.25, 3, 0, 3)
        TweenService:Create(tab_indicator, TweenInfo.new(0.45, SMOOTH, DIR_OUT), {Position = target_pos}):Play()
        for i, btn in ipairs(tab_btns) do
            local target_color = i == index and DEEP_BROWN or BROWN
            TweenService:Create(btn, TweenInfo.new(0.35, SOFT, DIR_OUT), {TextColor3 = target_color}):Play()
        end
    end

    for i = 1, 4 do
        local tb = Instance.new("TextButton")
        tb.Size = UDim2.new(0.25, 0, 1, 0)
        tb.Position = UDim2.new((i-1) * 0.25, 0, 0, 0)
        tb.BackgroundTransparency = 1
        tb.Text = tab_names[i]
        tb.TextColor3 = i == 1 and DEEP_BROWN or BROWN
        tb.TextSize = 14
        tb.Font = Enum.Font.GothamBold
        tb.AutoButtonColor = false
        tb.ZIndex = 6
        tb.Parent = tab_bar
        tb.MouseButton1Click:Connect(function() switch_tab(i) end)
        table.insert(tab_btns, tb)
    end

    get_or_create_scroll(1).Visible = true

    local drag_main = false
    local ds_main, sp_main
    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            drag_main = true ds_main = input.Position sp_main = main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then drag_main = false end
            end)
        end
    end)
    header.InputChanged:Connect(function(input)
        if drag_main and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - ds_main
            main.Position = UDim2.new(sp_main.X.Scale, sp_main.X.Offset + d.X, sp_main.Y.Scale, sp_main.Y.Offset + d.Y)
        end
    end)

    local min = false
    min_btn.MouseButton1Click:Connect(function()
        min = not min
        if min then
            content.Visible = false
            tab_bar.Visible = false
            TweenService:Create(main, TweenInfo.new(0.45, SMOOTH, DIR_OUT), {Size = UDim2.new(0, FRAME_W, 0, 60)}):Play()
            min_btn.Text = "+"
        else
            content.Visible = true
            tab_bar.Visible = true
            TweenService:Create(main, TweenInfo.new(0.45, SMOOTH, DIR_OUT), {Size = UDim2.new(0, FRAME_W, 0, FRAME_H)}):Play()
            min_btn.Text = "−"
        end
    end)

    -- force scroll canvases to compute AFTER everything is built
    task.wait(0.25)
    for _, sf in pairs(scroll_frames) do
        local lyt = sf:FindFirstChildOfClass("UIListLayout")
        if lyt then
            sf.CanvasSize = UDim2.new(0, 0, 0, lyt.AbsoluteContentSize.Y + 30)
        end
    end
    scroll_frames[1].Visible = true

    ui_flow_active = false
    notify("🍮 livyii.silent", "loaded for " .. platform .. " ✨", 4)
end

task.spawn(function()
    local ok, err = pcall(function()
        local outline_lines = {}
        local crosshair_lines = {}
        for i = 1, 4 do
            local line = Drawing.new("Line")
            line.Thickness = cfg.CrosshairOutlineThickness
            line.Color = BROWN
            line.Transparency = 1
            line.Visible = false
            table.insert(outline_lines, line)
        end
        for i = 1, 4 do
            local line = Drawing.new("Line")
            line.Thickness = cfg.CrosshairThickness
            line.Color = CREAM
            line.Transparency = 1
            line.Visible = false
            table.insert(crosshair_lines, line)
        end

        local fov_circle = Drawing.new("Circle")
        fov_circle.Thickness = cfg.FOVCircleThickness
        fov_circle.NumSides = 60
        fov_circle.Radius = cfg.FOVRadius
        fov_circle.Filled = false
        fov_circle.Color = PINK
        fov_circle.Transparency = cfg.FOVCircleTransparency
        fov_circle.Visible = false

        local info_text = Drawing.new("Text")
        info_text.Size = 14
        info_text.Center = true
        info_text.Outline = true
        info_text.OutlineColor = Color3.new(0, 0, 0)
        info_text.Color = PINK
        info_text.Font = 2
        info_text.Visible = false

        local highlight = Instance.new("Highlight")
        highlight.FillColor = PINK
        highlight.OutlineColor = PINK
        highlight.FillTransparency = 0.6
        highlight.OutlineTransparency = 0
        highlight.Enabled = false
        pcall(function() highlight.Parent = game:GetService("CoreGui") end)

        local tracer = Drawing.new("Line")
        tracer.Thickness = 1.5
        tracer.Color = PINK
        tracer.Transparency = 1
        tracer.Visible = false

        local rotation_angle = 0
        local function rotate_vector(v, angle)
            local c = math.cos(angle)
            local s = math.sin(angle)
            return Vector2.new(v.X * c - v.Y * s, v.X * s + v.Y * c)
        end

        local function is_knocked(char)
            local be = char and char:FindFirstChild("BodyEffects")
            if be then
                local ko = be:FindFirstChild("K.O")
                return ko and ko.Value
            end
            return false
        end

        local function is_holding_tool()
            if not cfg.RequireTool then return true end
            local char = LocalPlayer.Character
            if not char then return false end
            return char:FindFirstChildOfClass("Tool") ~= nil
        end

        local function is_visible(target_part)
            if not cfg.WallCheck then return true end
            local origin = cam.CFrame.Position
            local dir = target_part.Position - origin
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Blacklist
            params.FilterDescendantsInstances = {LocalPlayer.Character}
            local result = workspace:Raycast(origin, dir, params)
            if not result then return true end
            return result.Instance:IsDescendantOf(target_part.Parent)
        end

        local function get_aim_position(part)
            if cfg.AutoPredictionEnabled then
                local vel = part.AssemblyLinearVelocity or part.Velocity or Vector3.new(0,0,0)
                return part.Position + vel * cfg.PredictionStrength
            end
            return part.Position
        end

        local current_target_plr = nil

        local function get_target()
            if not cfg.Enabled then return nil end
            if not is_holding_tool() then return nil end
            local whitelist_active = has_active_whitelist()
            local aim_pos = cfg.UseCursorAim and UIS:GetMouseLocation() or Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
            local best_part, best_dist, best_plr = nil, cfg.FOVRadius, nil
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr == LocalPlayer then continue end
                if whitelist_active and not is_whitelisted(plr) then continue end
                local char = plr.Character
                if not char then continue end
                if cfg.KnockCheck and is_knocked(char) then continue end
                if cfg.TeamCheck and plr.Team == LocalPlayer.Team then continue end
                local part = char:FindFirstChild(cfg.HitPart) or char:FindFirstChild("HumanoidRootPart")
                if not part then continue end
                if not is_visible(part) then continue end
                local target_pos = get_aim_position(part)
                local sp, on_screen = cam:WorldToViewportPoint(target_pos)
                if not on_screen then continue end
                local dist = (Vector2.new(sp.X, sp.Y) - aim_pos).Magnitude
                if dist < best_dist then
                    best_dist = dist
                    best_part = part
                    best_plr = plr
                end
            end
            current_target_plr = best_plr
            return best_part
        end

        local function get_crosshair_draw_position()
            if cfg.FollowTarget then
                local target = get_target()
                if target then
                    local sp, on_screen = cam:WorldToViewportPoint(get_aim_position(target))
                    if on_screen then return Vector2.new(sp.X, sp.Y) end
                end
            end
            return cfg.CrosshairPosition
        end

        local function update_crosshair()
            if ui_flow_active or not cfg.CrosshairEnabled then
                for _, l in ipairs(outline_lines) do l.Visible = false end
                for _, l in ipairs(crosshair_lines) do l.Visible = false end
                return
            end
            local pos = get_crosshair_draw_position()
            local size = cfg.CrosshairSize
            local gap = cfg.CrosshairGap
            local tS = Vector2.new(0, -gap - size); local tE = Vector2.new(0, -gap)
            local bS = Vector2.new(0, gap); local bE = Vector2.new(0, gap + size)
            local lS = Vector2.new(-gap - size, 0); local lE = Vector2.new(-gap, 0)
            local rS = Vector2.new(gap, 0); local rE = Vector2.new(gap + size, 0)
            if cfg.CrosshairRotationEnabled then
                tS = rotate_vector(tS, rotation_angle); tE = rotate_vector(tE, rotation_angle)
                bS = rotate_vector(bS, rotation_angle); bE = rotate_vector(bE, rotation_angle)
                lS = rotate_vector(lS, rotation_angle); lE = rotate_vector(lE, rotation_angle)
                rS = rotate_vector(rS, rotation_angle); rE = rotate_vector(rE, rotation_angle)
            end
            local cross_color = Color3.fromHSV((cfg.CrosshairHue or 0) / 360, 1, 1)
            if cfg.CrosshairOutlineEnabled then
                outline_lines[1].From = pos + tS; outline_lines[1].To = pos + tE
                outline_lines[2].From = pos + bS; outline_lines[2].To = pos + bE
                outline_lines[3].From = pos + lS; outline_lines[3].To = pos + lE
                outline_lines[4].From = pos + rS; outline_lines[4].To = pos + rE
                for _, l in ipairs(outline_lines) do
                    l.Color = BROWN
                    l.Thickness = cfg.CrosshairOutlineThickness
                    l.Visible = true
                end
            else
                for _, l in ipairs(outline_lines) do l.Visible = false end
            end
            crosshair_lines[1].From = pos + tS; crosshair_lines[1].To = pos + tE
            crosshair_lines[2].From = pos + bS; crosshair_lines[2].To = pos + bE
            crosshair_lines[3].From = pos + lS; crosshair_lines[3].To = pos + lE
            crosshair_lines[4].From = pos + rS; crosshair_lines[4].To = pos + rE
            for _, l in ipairs(crosshair_lines) do
                l.Color = cross_color
                l.Thickness = cfg.CrosshairThickness
                l.Visible = true
            end
        end

        RunService.RenderStepped:Connect(function(dt)
            if ui_flow_active then
                fov_circle.Visible = false
                tracer.Visible = false
                info_text.Visible = false
                highlight.Enabled = false
                return
            end

            if cfg.UseCursorAim then
                local mp = UIS:GetMouseLocation()
                cfg.CrosshairPosition = Vector2.new(math.clamp(mp.X, 0, cam.ViewportSize.X), math.clamp(mp.Y, 0, cam.ViewportSize.Y))
            end
            if cfg.CrosshairRotationEnabled then
                rotation_angle = rotation_angle + cfg.CrosshairRotationSpeed * dt
            end
            update_crosshair()
            fov_circle.Position = cfg.CrosshairPosition
            fov_circle.Radius = cfg.FOVRadius
            fov_circle.Color = Color3.fromHSV((cfg.FOVHue or 0) / 360, 1, 1)
            fov_circle.Thickness = cfg.FOVCircleThickness
            fov_circle.Transparency = cfg.FOVCircleTransparency
            fov_circle.Visible = cfg.FOVCircleEnabled and cfg.Enabled

            local target_part = get_target()
            if target_part then
                local char = target_part.Parent
                if cfg.HighlightEnabled then
                    highlight.Adornee = char
                    highlight.Enabled = true
                else
                    highlight.Enabled = false
                end
                if cfg.TracerEnabled then
                    local sp, on_screen = cam:WorldToViewportPoint(get_aim_position(target_part))
                    if on_screen then
                        tracer.From = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y)
                        tracer.To = Vector2.new(sp.X, sp.Y)
                        tracer.Color = Color3.fromHSV((cfg.TracerHue or 0) / 360, 1, 1)
                        tracer.Visible = true
                    else
                        tracer.Visible = false
                    end
                else
                    tracer.Visible = false
                end
                if cfg.TargetInfoEnabled and current_target_plr then
                    local my_char = LocalPlayer.Character
                    local my_root = my_char and my_char:FindFirstChild("HumanoidRootPart")
                    if my_root then
                        local dist = math.floor((my_root.Position - target_part.Position).Magnitude)
                        info_text.Text = current_target_plr.Name .. " • " .. dist .. "m"
                        info_text.Position = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2 - 50)
                        info_text.Color = PINK
                        info_text.Visible = true
                    else
                        info_text.Visible = false
                    end
                else
                    info_text.Visible = false
                end
            else
                highlight.Enabled = false
                tracer.Visible = false
                info_text.Visible = false
            end
        end)

        if hookmetamethod and checkcaller then
            local old_index
            old_index = hookmetamethod(game, "__index", function(self, key)
                if not checkcaller() and cfg.Enabled and self == mouse and not ui_flow_active then
                    if key == "Hit" then
                        local target = get_target()
                        if target then return CFrame.new(get_aim_position(target)) end
                    elseif key == "Target" then
                        local target = get_target()
                        if target then return target end
                    end
                end
                return old_index(self, key)
            end)
        end

        pcall(function()
            local modules = ReplicatedStorage:FindFirstChild("Modules")
            if modules then
                local gun_handler = modules:FindFirstChild("GunHandler")
                if gun_handler then
                    local handler = require(gun_handler)
                    if handler and handler.getAim then
                        local old_get_aim = handler.getAim
                        handler.getAim = function(origin, max_dist)
                            if not cfg.Enabled or not is_holding_tool() or ui_flow_active then return old_get_aim(origin, max_dist) end
                            if cfg.RevolverBypass then
                                local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
                                if tool and tool.Name == "[Revolver]" then return old_get_aim(origin, max_dist) end
                            end
                            local target = get_target()
                            if target then
                                local aim_pos = get_aim_position(target)
                                return (aim_pos - origin).Unit, math.min((aim_pos - origin).Magnitude, max_dist or 200)
                            end
                            return old_get_aim(origin, max_dist)
                        end
                    end
                end
            end
        end)

        local drag_cross = false
        local drag_offset = Vector2.new(0, 0)
        UIS.TouchBegan:Connect(function(touch)
            if ui_flow_active then return end
            if cfg.FollowTarget and get_target() then return end
            if cfg.UseCursorAim then return end
            local pos = touch.Position
            local dist = (pos - cfg.CrosshairPosition).Magnitude
            if dist <= cfg.FOVRadius + 20 then
                drag_cross = true
                drag_offset = pos - cfg.CrosshairPosition
            end
        end)
        UIS.TouchMoved:Connect(function(touch)
            if drag_cross then cfg.CrosshairPosition = touch.Position - drag_offset end
        end)
        UIS.TouchEnded:Connect(function() drag_cross = false end)
        UIS.InputBegan:Connect(function(input, gp)
            if gp then return end
            if ui_flow_active then return end
            if cfg.FollowTarget and get_target() then return end
            if cfg.UseCursorAim then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                local pos = Vector2.new(input.Position.X, input.Position.Y)
                local dist = (pos - cfg.CrosshairPosition).Magnitude
                if dist <= cfg.FOVRadius + 20 then
                    drag_cross = true
                    drag_offset = pos - cfg.CrosshairPosition
                end
            end
        end)
        UIS.InputChanged:Connect(function(input)
            if drag_cross and input.UserInputType == Enum.UserInputType.MouseMovement then
                cfg.CrosshairPosition = Vector2.new(input.Position.X, input.Position.Y) - drag_offset
            end
        end)
        UIS.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then drag_cross = false end
        end)

        UIS.InputBegan:Connect(function(input, gp)
            if gp then return end
            if input.KeyCode == cfg.ToggleKey then
                cfg.Enabled = not cfg.Enabled
                notify("🍮 silent aim", cfg.Enabled and "enabled ✨" or "disabled 🌸", 2)
