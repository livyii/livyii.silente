    loadingLabel.Text = "loading.."

    local rotationAngle = 0
    local function rotateVector(v, angle)
        local cos = math.cos(angle)
        local sin = math.sin(angle)
        return Vector2.new(v.X * cos - v.Y * sin, v.X * sin + v.Y * cos)
    end

    local function isKnocked(char)
        local be = char and char:FindFirstChild("BodyEffects")
        if be then
            local ko = be:FindFirstChild("K.O")
            return ko and ko.Value
        end
        return false
    end

    local function isHoldingTool()
        if not cfg.RequireTool then return true end
        local char = LocalPlayer.Character
        if not char then return false end
        return char:FindFirstChildOfClass("Tool") ~= nil
    end

    local function getTarget()
        if not isHoldingTool() then return nil end
        local aimPos = cfg.CrosshairPosition
        local bestPart, bestDist = nil, cfg.FOVRadius
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr == LocalPlayer then continue end
            local char = plr.Character
            if not char then continue end
            if isKnocked(char) then continue end
            if cfg.TeamCheck and plr.Team == LocalPlayer.Team then continue end
            local part = char:FindFirstChild(cfg.HitPart) or char:FindFirstChild("HumanoidRootPart")
            if not part then continue end
            local sp, onScreen = cam:WorldToViewportPoint(part.Position)
            if not onScreen then continue end
            local dist = (Vector2.new(sp.X, sp.Y) - aimPos).Magnitude
            if dist < bestDist then
                bestDist = dist
                bestPart = part
            end
        end
        return bestPart
    end

    local function getCrosshairDrawPosition()
        if cfg.FollowTarget then
            local target = getTarget()
            if target then
                local sp, onScreen = cam:WorldToViewportPoint(target.Position)
                if onScreen then
                    return Vector2.new(sp.X, sp.Y)
                end
            end
        end
        return cfg.CrosshairPosition
    end

    local function updateCrosshair()
        if not cfg.CrosshairEnabled then
            for _, l in ipairs(outlineLines) do l.Visible = false end
            for _, l in ipairs(crosshairLines) do l.Visible = false end
            return
        end
        local pos = getCrosshairDrawPosition()
        local size = cfg.CrosshairSize
        local gap = cfg.CrosshairGap
        local topStart = Vector2.new(0, -gap - size)
        local topEnd   = Vector2.new(0, -gap)
        local botStart = Vector2.new(0, gap)
        local botEnd   = Vector2.new(0, gap + size)
        local leftStart = Vector2.new(-gap - size, 0)
        local leftEnd   = Vector2.new(-gap, 0)
        local rightStart = Vector2.new(gap, 0)
        local rightEnd   = Vector2.new(gap + size, 0)
        if cfg.CrosshairRotationEnabled then
            topStart = rotateVector(topStart, rotationAngle)
            topEnd   = rotateVector(topEnd, rotationAngle)
            botStart = rotateVector(botStart, rotationAngle)
            botEnd   = rotateVector(botEnd, rotationAngle)
            leftStart = rotateVector(leftStart, rotationAngle)
            leftEnd   = rotateVector(leftEnd, rotationAngle)
            rightStart = rotateVector(rightStart, rotationAngle)
            rightEnd   = rotateVector(rightEnd, rotationAngle)
        end
        local crossColor = Color3.fromHSV(cfg.CrosshairHue / 360, 1, 1)
        if cfg.CrosshairOutlineEnabled then
            outlineLines[1].From = pos + topStart
            outlineLines[1].To   = pos + topEnd
            outlineLines[2].From = pos + botStart
            outlineLines[2].To   = pos + botEnd
            outlineLines[3].From = pos + leftStart
            outlineLines[3].To   = pos + leftEnd
            outlineLines[4].From = pos + rightStart
            outlineLines[4].To   = pos + rightEnd
            for _, l in ipairs(outlineLines) do
                l.Color = cfg.CrosshairOutlineColor
                l.Thickness = cfg.CrosshairOutlineThickness
                l.Visible = true
            end
        else
            for _, l in ipairs(outlineLines) do l.Visible = false end
        end
        crosshairLines[1].From = pos + topStart
        crosshairLines[1].To   = pos + topEnd
        crosshairLines[2].From = pos + botStart
        crosshairLines[2].To   = pos + botEnd
        crosshairLines[3].From = pos + leftStart
        crosshairLines[3].To   = pos + leftEnd
        crosshairLines[4].From = pos + rightStart
        crosshairLines[4].To   = pos + rightEnd
        for _, l in ipairs(crosshairLines) do
            l.Color = crossColor
            l.Thickness = cfg.CrosshairThickness
            l.Visible = true
        end
    end
